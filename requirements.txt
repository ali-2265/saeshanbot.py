import os

from telegram import InlineKeyboardButton, InlineKeyboardMarkup, Update
from telegram.ext import (
    Application,
    CallbackQueryHandler,
    CommandHandler,
    ContextTypes,
)

# ============================================================
# Logging

    level=logging.INFO,
    format="%(asctime)s - %(levelname)s - %(message)s",
)

# ============================================================
# Main Menu
# ============================================================

def main_menu() -> InlineKeyboardMarkup:
    keyboard = [
        [
            InlineKeyboardButton(
                "🔐 Session Information",
                callback_data="session",
            )
        ],
        [
            InlineKeyboardButton(
                "❓ Help",
                callback_data="help",
            ),
            InlineKeyboardButton(
                "ℹ️ About",
                callback_data="about",
            ),
        ],
    ]

    return InlineKeyboardMarkup(keyboard)


# ============================================================
# Back Button
# ============================================================

def back_button() -> InlineKeyboardMarkup:
    return InlineKeyboardMarkup(
        [
            [
                InlineKeyboardButton(
                    "⬅️ Back",
                    callback_data="home",
                )
            ]
        ]
    )


# ============================================================
# /start
# ============================================================

async def start_command(
    update: Update,
    context: ContextTypes.DEFAULT_TYPE,
) -> None:

    if not update.message:
        return

    text = (
        "👋 Welcome!\n\n"
        "This is a simple and secure Telegram bot.\n\n"
        "Please select an option below:"
    )

    await update.message.reply_text(
        text,
        reply_markup=main_menu(),
    )


# ============================================================
# /help
# ============================================================

async def help_command(
    update: Update,
    context: ContextTypes.DEFAULT_TYPE,
) -> None:

    if not update.message:
        return

    text = (
        "❓ Help\n\n"
        "/start — Open the main menu\n"
        "/help — Show this help message\n"
        "/about — Information about the bot\n\n"
        "Security:\n"
        "Never share Telegram login codes, passwords, "
        "or Session Strings with anyone."
    )

    await update.message.reply_text(
        text,
        reply_markup=main_menu(),
    )


# ============================================================
# /about
# ============================================================

async def about_command(
    update: Update,
    context: ContextTypes.DEFAULT_TYPE,
) -> None:

    if not update.message:
        return

    text = (
        "ℹ️ About\n\n"
        "A lightweight Telegram bot built with Python "
        "and the Telegram Bot API.\n\n"
        "The bot does not collect or store Telegram "
        "account login information."
    )

    await update.message.reply_text(
        text,
        reply_markup=main_menu(),
    )


# ============================================================
# Button Handler
# ============================================================

async def button_handler(
    update: Update,
    context: ContextTypes.DEFAULT_TYPE,
) -> None:

    query = update.callback_query

    if not query:
        return

    await query.answer()

    try:
        if query.data == "home":

            text = (
                "👋 Welcome!\n\n"
                "This is a simple and secure Telegram bot.\n\n"
                "Please select an option below:"
            )

            await query.edit_message_text(
                text=text,
                reply_markup=main_menu(),
            )

        elif query.data == "session":

            text = (
                "🔐 Session Information\n\n"
                "A Telegram Session String is sensitive "
                "account information.\n\n"
                "Never share a Session String with anyone.\n\n"
                "This bot does NOT:\n"
                "• Request your phone number\n"
                "• Request Telegram login codes\n"
                "• Request your two-step verification password\n"
                "• Request, generate, or store Session Strings\n"
                "• Store login information"
            )

            await query.edit_message_text(
                text=text,
                reply_markup=back_button(),
            )

        elif query.data == "help":

            text = (
                "❓ Help\n\n"
                "/start — Open the main menu\n"
                "/help — Show help information\n"
                "/about — About this bot\n\n"
                "Keep your Telegram account credentials "
                "and Session String private."
            )

            await query.edit_message_text(
                text=text,
                reply_markup=back_button(),
            )

        elif query.data == "about":

            text = (
                "ℹ️ About\n\n"
                "This bot is built with Python and "
                "the Telegram Bot API.\n\n"
                "It is designed with privacy and security "
                "in mind and does not store user login data."
            )

            await query.edit_message_text(
                text=text,
                reply_markup=back_button(),
            )

    except Exception:
        logger.exception("An error occurred while handling a button.")


# ============================================================
# Error Handler
# ============================================================

async def error_handler(
    update: object,
    context: ContextTypes.DEFAULT_TYPE,
) -> None:

    logger.exception(
        "Unhandled bot exception.",
        exc_info=context.error,
    )


# ============================================================
# Main
# ============================================================

def main() -> None:

    application = (
        Application.builder()
        .token(BOT_TOKEN)
        .build()
    )

    application.add_handler(
        CommandHandler("start", start_command)
    )

    application.add_handler(
        CommandHandler("help", help_command)
    )

    application.add_handler(
        CommandHandler("about", about_command)
    )

    application.add_handler(
        CallbackQueryHandler(button_handler)
    )

    application.add_error_handler(error_handler)

    logger.info("Bot is running successfully.")

    application.run_polling(
        allowed_updates=Update.ALL_TYPES
    )


if __name__ == "__main__":
    main()
